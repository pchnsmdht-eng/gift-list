async function send(val, count) { const url = `https://api.github.com/repos/${USER}/${REPO}/contents/${FILE}`; try { // 1. GETリクエストを送り、最新のSHAを取得 const res = await fetch(`${url}?t=${Date.now()}`, { headers: { "Authorization": `token ${T}`, "Accept": "application/vnd.github.v3+json" } }); if (!res.ok) { addLog(`❌ 取得失敗: ${res.status}`); return; } const data = await res.json(); // 2. 更新実行 (Bodyの内容をシンプルに) const updateRes = await fetch(url, { method: "PUT", headers: { "Authorization": `token ${T}`, "Content-Type": "application/json" }, body: JSON.stringify({ message: "update speed", content: btoa(`${val},${count}`), sha: data.sha }) }); const result = await updateRes.json(); if (updateRes.ok) { addLog(`✅ 送信成功: 速度[${val}]`); } else { addLog(`❌ 失敗: ${result.message}`); // ここにGitHubからの拒否理由が出る } } catch (e) { addLog(`❌ システムエラー: ${e.message}`); } }
