async function send(val, count) { const url = `https://api.github.com/repos/${USER}/${REPO}/contents/${FILE}`; try { // 1. まず現在のファイル情報を取得 const res = await fetch(`${url}?t=${Date.now()}`, { headers: { "Authorization": `token ${T}`, "Cache-Control": "no-cache" } }); // ファイルが存在しない場合は処理を止める（手動で空のspeed.txtを作成しておく必要があります） if (!res.ok) throw new Error("ファイルが見つかりません"); const data = await res.json(); // 2. 文字列をBase64に正しく変換 const contentStr = `${val},${count}`; const encodedContent = btoa(unescape(encodeURIComponent(contentStr))); // 3. 更新実行 const updateRes = await fetch(url, { method: "PUT", headers: { "Authorization": `token ${T}`, "Content-Type": "application/json" }, body: JSON.stringify({ message: "update speed", content: encodedContent, sha: data.sha }) }); if (updateRes.ok) { addLog(`送信成功: 速度[${val}]`); } else { const err = await updateRes.json(); // ここで表示されるエラーメッセージが原因を特定する鍵です addLog(`❌ 失敗: ${err.message || '権限エラーの可能性'}`); } } catch (e) { addLog(`❌ 通信エラー: ${e.message}`); } }
