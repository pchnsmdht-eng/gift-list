// PUTリクエスト前の修正ロジック async function send(val, count) { const url = `https://api.github.com/repos/${USER}/${REPO}/contents/${FILE}`; try { // 1. まず現在のリポジトリの状態を取得（SHAを確実に取得） const res = await fetch(`${url}?t=${Date.now()}`, { headers: { "Authorization": `token ${T}`, "Cache-Control": "no-cache" } }); const data = await res.json(); // 2. 更新実行 const updateRes = await fetch(url, { method: "PUT", headers: { "Authorization": `token ${T}`, "Content-Type": "application/json" }, body: JSON.stringify({ message: "update speed", content: btoa(`${val},${count}`), sha: data.sha // ここで取得した最新のSHAを確実に渡す }) }); if (updateRes.ok) { addLog(`送信成功: 速度[${val}]`); } else { const err = await updateRes.json(); addLog(`❌ 失敗: ${err.message}`); // 詳細なエラーを表示 } } catch (e) { addLog(`❌ 通信エラー`); } }
